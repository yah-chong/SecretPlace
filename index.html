function buildBBox(lat, lng, radiusMeters) { const latDelta = radiusMeters / 111320; const lngDelta = radiusMeters / (111320 * Math.cos(lat * Math.PI / 180) || 1); return { south: (lat - latDelta).toFixed(5), north: (lat + latDelta).toFixed(5), west: (lng - lngDelta).toFixed(5), east: (lng + lngDelta).toFixed(5) }; } function getExtendedOverpassFilter(supplyKey, radiusMeters, lat, lng) { const r = String(radiusMeters); const la = String(lat); const lo = String(lng); if (supplyKey === 'fuel') { return ` nwr(around:${r},${la},${lo})["amenity"~"fuel|charging_station"]; nwr(around:${r},${la},${lo})["shop"="fuel"]; nwr(around:${r},${la},${lo})["brand"~"中油|CPC|台塑|Formosa|全國|山隆|台亞",i]; nwr(around:${r},${la},${lo})["operator"~"中油|CPC|台塑|Formosa|全國|山隆|台亞",i]; `; } if (supplyKey === 'food') { return ` nwr(around:${r},${la},${lo})["amenity"~"restaurant|cafe|fast_food|food_court|ice_cream"]; nwr(around:${r},${la},${lo})["shop"~"bakery|tea|coffee|beverages|pastry|confectionery|convenience|supermarket"]; `; } const baseCfg = SUPPLY_CONFIG[supplyKey]; return baseCfg.overpassFilter .replace(/RADIUS/g, r) .replace(/LAT/g, la) .replace(/LNG/g, lo);
